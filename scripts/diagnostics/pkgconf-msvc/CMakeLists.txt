cmake_minimum_required(VERSION 3.20)
project(pkgconf_native C)
set(CMAKE_C_STANDARD 11)
set(PKGCONF_SOURCE "" CACHE PATH "pkgconf 2.3.0 source")
set(sources argvsplit audit bsdstubs cache client dependency fileio fragment parser path personality pkg queue tuple)
set(files)
foreach(name IN LISTS sources)
  list(APPEND files "${PKGCONF_SOURCE}/libpkgconf/${name}.c")
endforeach()
add_executable(pkgconf ${files} "${PKGCONF_SOURCE}/cli/main.c" "${PKGCONF_SOURCE}/cli/getopt_long.c" "${PKGCONF_SOURCE}/cli/renderer-msvc.c")
target_include_directories(pkgconf PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PKGCONF_SOURCE}" "${PKGCONF_SOURCE}/libpkgconf" "${PKGCONF_SOURCE}/cli")
target_compile_definitions(pkgconf PRIVATE PKGCONFIG_IS_STATIC LIBPKGCONF_EXPORT _CRT_SECURE_NO_WARNINGS)
