cmake_minimum_required(VERSION 3.20)
project(photogogo_vidstab_backport C)
set(CMAKE_C_STANDARD 11)
set(VIDSTAB_SOURCE "" CACHE PATH "Pinned old vid.stab source directory")
if(NOT EXISTS "${VIDSTAB_SOURCE}/src/libvidstab.c")
  message(FATAL_ERROR "VIDSTAB_SOURCE must name the pinned source checkout")
endif()
find_package(OpenMP REQUIRED)
set(source_names frameinfo transformtype libvidstab transform transformfixedpoint
  motiondetect motiondetect_opt serialize localmotion2transform boxblur vsvector)
set(sources)
foreach(name IN LISTS source_names)
  list(APPEND sources "${VIDSTAB_SOURCE}/src/${name}.c")
endforeach()
add_library(vidstab STATIC ${sources} "${VIDSTAB_SOURCE}/src/orc/motiondetectorc.c")
target_include_directories(vidstab PUBLIC "${VIDSTAB_SOURCE}/src")
target_compile_definitions(vidstab PUBLIC USE_OMP DISABLE_ORC _USE_MATH_DEFINES)
target_link_libraries(vidstab PUBLIC OpenMP::OpenMP_C)
if(MSVC)
  target_compile_definitions(vidstab PRIVATE _CRT_SECURE_NO_WARNINGS USE_SSE2)
  target_compile_options(vidstab PRIVATE /W3)
else()
  target_link_libraries(vidstab PUBLIC m)
endif()
add_executable(backport-test backport-test.c)
target_link_libraries(backport-test PRIVATE vidstab)
add_executable(binary-header-test binary-header-test.c)
target_link_libraries(binary-header-test PRIVATE vidstab)
install(TARGETS vidstab ARCHIVE DESTINATION lib)
install(DIRECTORY "${VIDSTAB_SOURCE}/src/" DESTINATION include/vid.stab FILES_MATCHING PATTERN "*.h")
